"""Unit tests for hello.py (Minimal strategy, express scope).

One verifiable test per requirement; tests invoke the script as a
subprocess (`python3 hello.py`) and assert on stdout, stderr, and exit
code — the narrowest level that verifies each requirement.
"""

import subprocess
import unittest
from pathlib import Path

WORKSPACE_ROOT = Path(__file__).resolve().parent
HELLO = WORKSPACE_ROOT / "hello.py"


def run_hello() -> subprocess.CompletedProcess:
    """Run `python3 hello.py` at the workspace root, capturing output."""
    return subprocess.run(
        ["python3", "hello.py"],
        cwd=WORKSPACE_ROOT,
        capture_output=True,
        text=True,
    )


class TestHello(unittest.TestCase):
    def test_hello_file_exists(self):
        """FR1: hello.py exists at the workspace root and is a regular file."""
        self.assertTrue(HELLO.exists(), "hello.py must exist at workspace root")
        self.assertTrue(HELLO.is_file(), "hello.py must be a regular file")

    def test_prints_ok(self):
        """FR1.1 (happy path): stdout is exactly 'ok\\n'."""
        result = run_hello()
        self.assertEqual("ok\n", result.stdout)

    def test_clean_exit(self):
        """FR1.2: stderr is empty and exit code is 0."""
        result = run_hello()
        self.assertEqual("", result.stderr)
        self.assertEqual(0, result.returncode)

    def test_file_is_single_print(self):
        """FR1.3: file content is exactly the single statement print("ok")."""
        self.assertEqual('print("ok")\n', HELLO.read_text())


if __name__ == "__main__":
    unittest.main()
